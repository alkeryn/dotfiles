cmake_minimum_required(VERSION 3.25)

# Prefer Clang; an explicit -DCMAKE_CXX_COMPILER still wins.
if(NOT DEFINED CMAKE_CXX_COMPILER)
    find_program(CLANGXX clang++ REQUIRED)
    set(CMAKE_CXX_COMPILER "${CLANGXX}" CACHE FILEPATH "C++ compiler")
endif()
project(hypr_extras VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()

find_package(PkgConfig REQUIRED)
pkg_check_modules(HYPRLAND REQUIRED IMPORTED_TARGET hyprland)
add_library(extras_options INTERFACE)
target_link_libraries(extras_options INTERFACE PkgConfig::HYPRLAND ${CMAKE_DL_LIBS})
target_compile_options(extras_options INTERFACE -Wall -Wextra -Wpedantic)
target_link_options(extras_options INTERFACE -Wl,--as-needed)
if(CMAKE_CXX_COMPILER_ID MATCHES "Clang")
    target_compile_options(extras_options INTERFACE -stdlib=libstdc++ -Wno-return-type-c-linkage)
    target_link_options(extras_options INTERFACE -stdlib=libstdc++)
endif()
# PluginAPI.hpp needs the dependency versions to construct the ABI string.
foreach(dep IN ITEMS aquamarine hyprutils hyprgraphics hyprcursor hyprlang)
    pkg_check_modules(${dep} REQUIRED ${dep})
    string(TOUPPER "${dep}" upper)
    target_compile_definitions(extras_options INTERFACE ${upper}_VERSION="${${dep}_VERSION}")
endforeach()

set(extras_sources src/plugin.cpp src/hooks.cpp src/focus.cpp)
add_library(hypr_extras SHARED ${extras_sources})
target_link_libraries(hypr_extras PRIVATE extras_options)
# Keep Hyprland's inline globals interposable (notably Log::logger).
set_target_properties(hypr_extras PROPERTIES PREFIX "" VISIBILITY_INLINES_HIDDEN YES)

include(CTest)
if(BUILD_TESTING)
    add_executable(extras_test tests/extras_test.cpp src/hooks.cpp)
    target_link_libraries(extras_test PRIVATE extras_options)
    target_compile_options(extras_test PRIVATE -UNDEBUG -fno-optimize-sibling-calls)
    target_link_options(extras_test PRIVATE -rdynamic)
    add_dependencies(extras_test hypr_extras)
    add_test(NAME extras COMMAND extras_test $<TARGET_FILE:hypr_extras>)
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
    find_program(HYPRLAND_BINARY Hyprland REQUIRED)
    add_test(NAME binary_contract COMMAND ${Python3_EXECUTABLE}
        ${CMAKE_CURRENT_SOURCE_DIR}/tests/binary_contract.py ${HYPRLAND_BINARY})
endif()
